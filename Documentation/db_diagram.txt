Table patient_demographics {

  Patient_ID varchar [pk]

  Date_of_Birth date
  Age int
  Sex varchar

  Height_cm float
  Weight_kg float
  BMI float

  Neck_Circumference_cm float

  Family_History_OSA boolean

  Tobacco_Use boolean
  Alcohol_Use boolean
  Caffeine_Use boolean

  Shift_Work boolean
}

Table patients {

  Patient_ID varchar [pk, ref: > patient_demographics.Patient_ID]

  Consultation_Date date
  Clinic_Number varchar
  Referring_Physician varchar

  Insurance_Type varchar
  Payment_Type varchar
}

  Patient_ID varchar [pk, ref: > patient_demographics.Patient_ID]

  Snoring boolean
  Witnessed_Apneas boolean
  Nocturnal_Choking boolean
  Insomnia boolean
  Night_Awakenings boolean
  Non_Restorative_Sleep boolean
  Daytime_Sleepiness boolean
  Nasal_Obstruction boolean
  Dry_Mouth boolean
  Morning_Headaches boolean
  Jaw_Pain boolean
  Teeth_Grinding boolean
}

Table medical_history {

  Patient_ID varchar [pk, ref: > patient_demographics.Patient_ID]

  Hypertension boolean
  Diabetes boolean
  Depression boolean
  GERD boolean
  Cancer boolean

  Medication varchar
  Surgery varchar
  Allergies text
}

  Patient_ID varchar [pk, ref: > patient_demographics.Patient_ID]

  Insomnia boolean
  Parasomnias boolean

  Sleep_Position boolean
  MFT boolean

}

  Patient_ID varchar [pk, ref: > patient_demographics.Patient_ID]

  Epworth_Score int

  QoL_Score float

  SnoreLab_Score float
  SnoreLab_Epic float
}
